string validateAnswer(int len, string guess, string secret) {
  vector<int> arr(26);
  for (int i = 0; i < len; i++) {
    arr[secret[i]-'a']++;
  }

  string s(len, 'x');

  for (int i = 0; i < len; i++) {
    if (guess[i] == secret[i]) {
      arr[guess[i]-'a']--;
      s[i] = '.';
    }
  }

  for (int i = 0; i < len; i++) {
    if (guess[i] != secret[i] && arr[guess[i]-'a'] > 0) {
      arr[guess[i]-'a']--;
      s[i] = 'o';
    }
  }

  return s;
}
